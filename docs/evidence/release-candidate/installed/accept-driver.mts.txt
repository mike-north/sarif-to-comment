 * the installed-package suite of the earlier acceptance at 89826ad
 * (docs/evidence/release-acceptance/installed/accept-driver.mts.txt), steps

 * characters, the nesting of outcome Markdown, the projected diff with

 *
 * It exercises the INSTALLED package only — the `sarif-to-comment`

 *
 * Every step records its command, exit status, full stdout/stderr and the
 * checks made on them. Expected values come from the contracts
 * (docs/*-contract.md), not from earlier output.
 *
 * Run from a fresh clone at the candidate commit, with
 * TMPDIR=<clone>/scratch/rc/tmp so every fake host lives in scratch.

function redact(text: string): string {
  return text
    .split(TMP).join('<tmp>')
    .split(CONSUMER).join('<consumer>')
    .split(WT).join('<repo>')
    .split(os.homedir()).join('~');
}

interface ICheck { readonly name: string; readonly pass: boolean; readonly detail?: string }
interface IStep {
  readonly id: string;
  readonly feature: string;
  readonly title: string;
  readonly command: string;
  readonly status: number | null;
  readonly stdout: string;
  readonly stderr: string;
  readonly script?: string;
  inputs?: string;
  observed?: string;
  readonly checks: ICheck[];
  check(name: string, pass: boolean, detail?: unknown): boolean;
  json(): Json;
}
const STEPS: IStep[] = [];
const quote = (a: string): string => (/^[A-Za-z0-9_./:=@,+-]+$/.test(a) ? a : `'${a.replace(/'/g, `'\\''`)}'`);

function makeStep(feature: string, title: string, command: string, status: number | null, stdout: string, stderr: string, tokens: readonly string[], script?: string): IStep {
  const id = String(STEPS.length + 1).padStart(3, '0');
  const step: IStep = {
    id, feature, title, command, status, stdout, stderr, ...(script === undefined ? {} : { script }), checks: [],
    check(name, pass, detail) {
      this.checks.push({ name, pass, ...(pass || detail === undefined ? {} : { detail: redact(typeof detail === 'string' ? detail : JSON.stringify(detail)).slice(0, 2000) }) });
      return pass;
    },
    json() {
      return JSON.parse(this.stdout) as Json;
    },
  };
  for (const t of tokens) step.check('the token never appears in stdout or stderr', !stdout.includes(t) && !stderr.includes(t));
  STEPS.push(step);
  return step;
}

interface IHostRef { readonly host: FakeHttpGitHub }
interface ICliOptions { readonly world?: IHostRef | undefined; readonly token?: string | undefined; readonly env?: Record<string, string> | undefined; readonly cwd?: string | undefined }

function cli(feature: string, title: string, args: readonly string[], o: ICliOptions = {}): IStep {
  const hostEnv: Record<string, string> = o.world === undefined ? {} : {
    GH_TOKEN: o.token ?? '', FAKE_HTTP_GITHUB_DIR: o.world.host.dir, NODE_OPTIONS: `--require=${PRELOAD}`,
  };
  const env: NodeJS.ProcessEnv = { PATH: process.env['PATH'], HOME: TMP, ...hostEnv, ...(o.env ?? {}) };
  const sarifIndex = args.indexOf('--sarif');
  const inputPath = sarifIndex >= 0 ? args[sarifIndex + 1] : undefined;
  const inputs = inputPath !== undefined && fs.existsSync(inputPath) ? read(inputPath) : undefined;
  const r = spawnSync(BIN, args, { cwd: o.cwd ?? CONSUMER, env, encoding: 'utf8', timeout: 180_000 });
  const shownEnv = Object.entries({ ...hostEnv, ...(o.env ?? {}) }).map(([k, v]) => `${k}=${k === 'GH_TOKEN' ? '<fake-host-token>' : quote(v)}`);
  const command = redact([...shownEnv, 'sarif-to-comment', ...args.map(quote)].join(' '));
  const step = makeStep(feature, title, command, r.status, r.stdout, r.stderr, o.token === undefined ? [] : [o.token]);
  if (inputs !== undefined) step.inputs = inputs;
  if (o.world !== undefined) step.observed = observe(o.world.host);
  return step;
}

/** What the fake host holds after a command: every review request and every pull request (for the record). */
function observe(host: FakeHttpGitHub): string {
  const writes = nonGetWrites(host);
  const reviews = host.reviews().map((r) => ({ id: r.id, state: r.state, request: r.request }));
  const pulls = host.pulls().map((p) => ({ number: p.number, title: p.title, head: p.head, base: p.base, draft: p.draft, state: p.state, labels: p.labels, body: p.body }));
  return JSON.stringify({ writesSoFar: writes, reviews, pulls }, null, 2);
}

function lib(feature: string, title: string, name: string, source: string, o: ICliOptions = {}): IStep {
  const file = path.join(CONSUMER, `u8-${name}.mjs`);
  fs.writeFileSync(file, source);
  const hostEnv: Record<string, string> = o.world === undefined ? {} : {
    GH_TOKEN: o.token ?? '', FAKE_HTTP_GITHUB_DIR: o.world.host.dir, NODE_OPTIONS: `--require=${PRELOAD}`,
  };
  const env: NodeJS.ProcessEnv = { PATH: process.env['PATH'], HOME: TMP, ...hostEnv, ...(o.env ?? {}) };
  const r = spawnSync(process.execPath, [file], { cwd: CONSUMER, env, encoding: 'utf8', timeout: 180_000 });
  const shownEnv = Object.entries({ ...hostEnv, ...(o.env ?? {}) }).map(([k, v]) => `${k}=${k === 'GH_TOKEN' ? '<fake-host-token>' : quote(v.length > 200 ? `${v.slice(0, 60)}…(${String(v.length)} chars)` : v)}`);
  const step = makeStep(feature, title, redact([...shownEnv, 'node', `<consumer>/u8-${name}.mjs`].join(' ')), r.status, r.stdout, r.stderr, o.token === undefined ? [] : [o.token], source);
  if (o.world !== undefined) step.observed = observe(o.world.host);
  return step;
}

const codes = (doc: Json): unknown[] => ((doc['diagnostics'] ?? []) as Json[]).map((d) => d['code']);
const has = (s: string, part: string): boolean => s.includes(part);
const read = (p: string): string => fs.readFileSync(p, 'utf8');
const writeSarif = (dir: string, name: string, sarif: Json): string => {
  const file = path.join(dir, name);
  fs.writeFileSync(file, `${JSON.stringify(sarif, null, 2)}\n`);
  return file;
};
const workDir = (label: string): string => fs.mkdtempSync(path.join(TMP, `${label}-`));
const nonGetWrites = (host: FakeHttpGitHub): string[] => host.log().filter((r) => r.method !== 'GET' && !(r.method === 'POST' && r.path === '/graphql')).map((r) => `${r.method} ${r.path}`);

const fileText = (host: FakeHttpGitHub, pull: IStoredPull | undefined, file: string): string | null => {
  if (pull === undefined) return null;
  const b = host.fileOnBranch(pull.head, file);
  return b === null ? null : b.toString('utf8');
};
const dTarget = (w: D.IWorld, sarif: Json): string[] => D.target(w, sarif);
const dToken = D.TOKEN;
const opt = (w: D.IWorld): ICliOptions => ({ world: w, token: dToken });

// 14. The path rule for invisible characters (delivery policy §8.10; file-operation contract §2;
//     review presentation contract §4): a path holding U+200B or U+00A0 is refused in each of the
//     three forms that name a path in the review: a whole-file proposal, an alternative, an edit made by hand.
{
  const chars = [['U+200B', '​', '%E2%80%8B'], ['U+00A0', ' ', '%C2%A0']] as const;
  /** D's repository with extra files outside the diff, at the base and the head. */
  const withFiles = (extra: Record<string, readonly string[]>): IHttpRepository => {
    const r = D.repository();
    const snaps = r.snapshots;
    return { ...r, snapshots: { ...snaps, [D.BASE]: { ...snaps[D.BASE], ...extra }, [D.HEAD]: { ...snaps[D.HEAD], ...extra } } };
  };
  const diag = (s: IStep): Json[] => ((s.json()['diagnostics'] ?? []) as Json[]);
  for (const [name, ch, enc] of chars) {
    // a. A whole-file proposal (creation) under the defaults (fileOperations [manual]).
    const w1 = D.makeWorld();
    const created = `docs/x${enc}y.md`;
    const doc1 = D.document([D.result({ text: 'Add the page.', location: D.at(created, 1), operation: D.create(0) })], [D.created(created, 'x\n')]);
    const a = cli('path rule: whole-file proposal', `a creation whose path holds ${name}: refused (file-operation-path-unrepresentable)`, ['publish', ...dTarget(w1, doc1), '--state', w1.statePath, '--format', 'json'], opt(w1));
    a.check('exit 2, blocked', a.status === 2 && a.json()['status'] === 'blocked', a.stdout);
    a.check('one problem, file-operation-path-unrepresentable', JSON.stringify(codes(a.json())) === '["file-operation-path-unrepresentable"]', codes(a.json()));
    a.check(`its message names the character: "contains ${name}, which cannot be shown exactly"`, String(diag(a)[0]?.['message'] ?? '').includes(`contains ${name}, which cannot be shown exactly`), diag(a)[0]);
    a.check('nothing written, no state file', nonGetWrites(w1.host).length === 0 && !fs.existsSync(w1.statePath), nonGetWrites(w1.host));

    // b. An alternative on another file whose path holds the character.
    const altPath = `alt${ch}name.js`;
    const w2 = D.makeWorld(withFiles({ [altPath]: ['odd\n'] }));
    const doc2 = D.document([D.result({ text: 'Two ways.', location: D.at('README.md', 2), fixes: [D.lineFix('README.md', 2, 'The widget client.'), D.lineFix(`alt${enc}name.js`, 1, 'even')] })]);
    const b = cli('path rule: alternative', `an alternative on a file whose path holds ${name}: refused (alternative-path-unrepresentable)`, ['publish', ...dTarget(w2, doc2), '--state', w2.statePath, '--format', 'json'], opt(w2));
    b.check('exit 2, blocked', b.status === 2 && b.json()['status'] === 'blocked', b.stdout);
    b.check('one problem, alternative-path-unrepresentable', JSON.stringify(codes(b.json())) === '["alternative-path-unrepresentable"]', codes(b.json()));
    b.check(`its message says "the file path contains ${name}, which cannot be shown exactly"`, String(diag(b)[0]?.['message'] ?? '').includes(`the file path contains ${name}, which cannot be shown exactly`), diag(b)[0]);
    b.check('nothing written, no state file', nonGetWrites(w2.host).length === 0 && !fs.existsSync(w2.statePath), nonGetWrites(w2.host));

    // c. An edit made by hand (--edits review-body, strict) of a file whose path holds the character.
    const handPath = `hand${ch}name.txt`;
    const w3 = D.makeWorld(withFiles({ [handPath]: ['x\n'] }));
    const doc3 = D.document([D.result({ text: 'Change it.', location: D.at(`hand${enc}name.txt`, 1), fixes: [D.lineFix(`hand${enc}name.txt`, 1, 'y')] })]);
    const c = cli('path rule: edit made by hand', `\`--edits review-body\`: an edit of a file whose path holds ${name}: refused (delivery-unavailable)`, ['publish', ...dTarget(w3, doc3), '--edits', 'review-body', '--state', w3.statePath, '--format', 'json'], opt(w3));
    c.check('exit 2, blocked, delivery-unavailable', c.status === 2 && c.json()['status'] === 'blocked' && JSON.stringify(codes(c.json())) === '["delivery-unavailable"]', c.stdout);
    c.check(`the review-body obstacle ends "the file path contains ${name}, which cannot be shown exactly."`, String(diag(c)[0]?.['message'] ?? '').includes(`cannot be shown exactly in the review body: the file path contains ${name}, which cannot be shown exactly.`), diag(c)[0]);
    c.check('nothing written, no state file', nonGetWrites(w3.host).length === 0 && !fs.existsSync(w3.statePath), nonGetWrites(w3.host));
  }

  s.check('concurrency cancels an earlier waiting run for the same pull request', JSON.stringify(job?.['concurrency']) === JSON.stringify({ group: 'abandonment-cleanup-${{ github.event.pull_request.number }}', 'cancel-in-progress': true }), job?.['concurrency']);

  const al = spawnSync('actionlint', ['-no-color', file], { encoding: 'utf8' });
  const ver = spawnSync('actionlint', ['--version'], { encoding: 'utf8' });
  const a = makeStep('example workflow', `actionlint ${ver.stdout.split('\n')[0] ?? ''} on the example workflow`, 'actionlint -no-color <repo>/docs/examples/abandonment-cleanup.workflow.yml', al.status, al.stdout, al.stderr, []);
  a.check('actionlint is available and reports nothing (exit 0)', al.error === undefined && al.status === 0 && al.stdout === '', al.stdout + al.stderr + String(al.error ?? ''));

  const h = cli('example workflow', 'every option the workflow passes exists in the installed close-suggestion-prs --help', ['close-suggestion-prs', '--help']);
  h.check('--repo, --original, --if-abandoned and --owner me|all are in the installed help', h.status === 0 && ['--repo OWNER/REPO', '--original N', '--if-abandoned', '--owner me|all'].every((f) => h.stdout.includes(f)), h.stdout);
}

// Write the evidence
fs.mkdirSync(path.join(OUT, 'steps'), { recursive: true });
const slug = (t: string): string => t.toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/^-|-$/g, '').slice(0, 60);
const summary = STEPS.map((s) => {
  const file = `steps/${s.id}-${slug(s.title)}.txt`;
  const text = [
    `# ${s.id} ${s.title}`,
    `feature: ${s.feature}`,
    '',
    `$ ${s.command}`,
    `exit status: ${String(s.status)}`,
    '',
    'checks:',
    ...s.checks.map((c) => `  ${c.pass ? 'PASS' : 'FAIL'}  ${c.name}${c.detail === undefined ? '' : `\n        ${c.detail}`}`),
    ...(s.script === undefined ? [] : ['', '--- script', redact(s.script).trim()]),
    '',
    '--- stdout',
    redact(s.stdout),
    '--- stderr',
    redact(s.stderr),
    ...(s.inputs === undefined ? [] : ['--- input SARIF (as given to --sarif, before the command)', redact(s.inputs)]),
    ...(s.observed === undefined ? [] : ['--- fake host after the command (writes so far, review requests, pull requests)', redact(s.observed)]),
  ].join('\n');
  fs.writeFileSync(path.join(OUT, file), `${text.trimEnd()}\n`);
  return { id: s.id, feature: s.feature, title: s.title, command: s.command, exit: s.status, pass: s.checks.every((c) => c.pass), checks: s.checks.map((c) => ({ name: c.name, pass: c.pass })), file };
});
fs.writeFileSync(path.join(OUT, 'summary.json'), `${JSON.stringify(summary, null, 2)}\n`);
const failed = summary.filter((s) => !s.pass);
process.stdout.write(`${String(summary.length)} steps, ${String(summary.reduce((n, s) => n + s.checks.length, 0))} checks; failed steps: ${String(failed.length)}\n`);
for (const s of failed) process.stdout.write(`FAIL ${s.id} ${s.title}: ${s.checks.filter((c) => !c.pass).map((c) => c.name).join('; ')}\n`);
